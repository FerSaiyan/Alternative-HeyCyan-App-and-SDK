cmake_minimum_required(VERSION 3.22)
project(cyan_shared_gemma LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
# Same verified pin as the Eyevue upstream adapter. Generic next-token scoring
# needs no dedicated decision-model graph or experimental server endpoint.
if(CYAN_LLAMA_SOURCE_DIR)
    add_subdirectory("${CYAN_LLAMA_SOURCE_DIR}" llama-upstream)
else()
    include(FetchContent)
    FetchContent_Declare(llama_upstream
        URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/0c1e57098bba43ac29e6e3b677cdceebdd22334f
        URL_HASH SHA256=6a85ec945b57789c875e1af90e66109fbf136a3c79802d5ca0333b043e10407f)
    FetchContent_MakeAvailable(llama_upstream)
endif()
add_library(cyan_shared_gemma SHARED bridge.cpp)
target_link_libraries(cyan_shared_gemma PRIVATE llama llama-common)
if(ANDROID)
    target_link_libraries(cyan_shared_gemma PRIVATE android log)
else()
    if(CYAN_JNI_INCLUDE_DIR)
        # The NDK's VM-neutral jni.h also works for host contract smokes.
        # Isolate it so the host compiler still uses its own libc headers.
        configure_file("${CYAN_JNI_INCLUDE_DIR}/jni.h" "${CMAKE_CURRENT_BINARY_DIR}/jni/jni.h" COPYONLY)
        target_include_directories(cyan_shared_gemma PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/jni")
    else()
        find_package(JNI REQUIRED)
        target_include_directories(cyan_shared_gemma PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()
target_link_options(cyan_shared_gemma PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
