cmake_minimum_required(VERSION 3.22)
project(cyan_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# The shared-Gemma decision prototype (cyan_shared_gemma) needs llama-common chat templates.
if(CYAN_SHARED_GEMMA)
    set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
else()
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)

# Official llama.cpp + libmtmd; pin both revision and archive digest.
if(CYAN_LLAMA_SOURCE_DIR)
    set(llama_upstream_SOURCE_DIR "${CYAN_LLAMA_SOURCE_DIR}")
    add_subdirectory("${llama_upstream_SOURCE_DIR}" llama-upstream)
else()
    include(FetchContent)
    FetchContent_Declare(llama_upstream
        URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/0c1e57098bba43ac29e6e3b677cdceebdd22334f
        URL_HASH SHA256=6a85ec945b57789c875e1af90e66109fbf136a3c79802d5ca0333b043e10407f)
    FetchContent_MakeAvailable(llama_upstream)
endif()
add_subdirectory("${llama_upstream_SOURCE_DIR}/tools/mtmd" mtmd)
add_library(cyan_llama SHARED bridge.cpp)
target_link_libraries(cyan_llama PRIVATE llama mtmd)
if(ANDROID)
    target_link_libraries(cyan_llama PRIVATE android log)
else()
    # Host JNI smoke tests exercise the same bridge without an Android device.
    if(CYAN_JNI_INCLUDE_DIR)
        # The NDK's VM-neutral jni.h also works for host contract smokes.
        # Isolate it so the host compiler still uses its own libc headers.
        configure_file("${CYAN_JNI_INCLUDE_DIR}/jni.h" "${CMAKE_CURRENT_BINARY_DIR}/jni/jni.h" COPYONLY)
        set(JNI_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/jni")
    else()
        find_package(JNI REQUIRED)
    endif()
    target_include_directories(cyan_llama PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_options(cyan_llama PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")

# Opt-in shared-Gemma decision bridge (-PsharedGemmaPrototype=true): next-token label scoring.
if(CYAN_SHARED_GEMMA)
    add_library(cyan_shared_gemma SHARED shared_gemma_bridge.cpp)
    target_link_libraries(cyan_shared_gemma PRIVATE llama llama-common)
    if(ANDROID)
        target_link_libraries(cyan_shared_gemma PRIVATE android log)
    else()
        target_include_directories(cyan_shared_gemma PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
    target_link_options(cyan_shared_gemma PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
endif()
